# GlobalNews v2.0 重构动工指南

> **执行方式：** 按任务分发给子 agent 逐个执行（subagent-driven），每个任务完成后由主会话审查再派下一个。
> 步骤使用 `- [ ]` 复选框跟踪进度。

**目标：** 将现有单文件 RSS 聚合器升级为：权威信源 + 代理支持 + 内嵌定时调度 + 多级网页抓取 + PDF 附件归档 + Claude 风格日/夜双主题前端。

**架构：** Flask 单进程同时承载 Web 服务与周六 9 点调度线程；爬虫逻辑拆分为 `crawler/` 包（sources / fetch / detail / attachments / pipeline / render）；数据全量保留于 `news_data.json`，详情与附件落盘 `cache/` 与 `attachments/`。

**技术栈：** Python 3.13, Flask, feedparser, requests, schedule（复用）, deep-translator（复用）, trafilatura（新增）

**已确认决策（2026-09-02）：**
1. 代理 `http://127.0.0.1:7993`，境外源走代理、国内源强制直连
2. 取消 30 天月度清理与每栏目 30 条上限，**历史数据全量保留**
3. 附件仅保留单文件 20MB 上限，不设每次爬取总数上限
4. 多级爬取 depth=1（文章页），每分类 top N=10
5. 前端卡片点击 → 右侧详情抽屉（不再直接跳原网页）
6. 移除澎湃新闻、36氪（分类不符）；保留机器之心、China Daily

```python
# -*- coding: utf-8 -*-
"""RSS 抓取 + 代理管理。境外源走代理，国内源强制直连。"""
import json, os, re, hashlib, logging, time
from typing import Dict, List
import feedparser, requests

logger = logging.getLogger("crawler")

BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
CONFIG_FILE = os.path.join(BASE_DIR, "config.json")

UA = ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
      "(KHTML, like Gecko) Chrome/125.0.0.0 Safari/537.36")

def load_config() -> Dict:
    try:
        with open(CONFIG_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except (IOError, json.JSONDecodeError):
        return {}

def build_session(needs_proxy: bool, proxy: str) -> requests.Session:
    """needs_proxy=True → 走代理；False → 显式清空代理并禁用系统代理。"""
    s = requests.Session()
    s.headers.update({"User-Agent": UA, "Accept": "application/rss+xml, application/xml, text/xml, */*"})
    if needs_proxy and proxy:
        s.proxies = {"http": proxy, "https": proxy}
    else:
        s.proxies = {}
        s.trust_env = False   # 防系统代理劫持国内流量
    return s

def make_id(link: str) -> str:
    return hashlib.md5(link.strip().encode("utf-8")).hexdigest()
```

```python
# -*- coding: utf-8 -*-
"""逐源探测可用性：HTTP 状态 + RSS 是否解析出条目。"""
import concurrent.futures
from typing import Dict, List
import feedparser
from crawler.sources import RSS_SOURCES
from crawler.fetch import build_session, load_config

def check_all() -> List[Dict]:
    cfg = load_config()
    proxy = cfg.get("proxy", "")
    def _probe(src):
        s = build_session(src["needs_proxy"], proxy)
        try:
            r = s.get(src["url"], timeout=15)
            feed = feedparser.parse(r.content)
            n = len(feed.entries)
            return {**src, "status": r.status_code, "entries": n,
                    "ok": r.ok and n > 0}
        except Exception as e:
            return {**src, "status": 0, "entries": 0, "ok": False, "error": str(e)}
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
        return list(pool.map(_probe, RSS_SOURCES))

def print_report(results: List[Dict]) -> None:
    for r in results:
        flag = "OK " if r["ok"] else "FAIL"
        print(f"[{flag}] {r['name']:<24} HTTP {r['status']:<4} {r['entries']} 条"
              + (f"  {r.get('error','')}" if not r["ok"] else ""))
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Global News v2 — Flask + 内嵌调度器，唯一常驻入口。"""
import os, sys, json, time, threading, logging
from datetime import datetime, timedelta

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from flask import Flask, jsonify, send_from_directory

try:
    import schedule as schedule_lib
    HAS_SCHEDULE = True
except ImportError:
    HAS_SCHEDULE = False

from crawler import run_crawl          # Task 6 起从包导入；此前暂从旧 crawler.py

app = Flask(__name__)
BASE_DIR = os.path.dirname(os.path.abspath(__file__))

_crawl_lock = threading.Lock()          # 手动 + 定时共用互斥
_crawl_in_progress = False
_crawl_result = None

def _safe_run_crawl(trigger: str) -> None:
    """互斥保护下执行爬取。trigger: 'manual' | 'scheduled' | 'startup'"""
    global _crawl_in_progress, _crawl_result
    if not _crawl_lock.acquire(blocking=False):
        logging.getLogger("web").info("已有爬取进行中，跳过(%s)", trigger)
        return
    _crawl_in_progress = True
    _crawl_result = None
    try:
        run_crawl()
        _crawl_result = {"ok": True, "trigger": trigger,
                         "finished_at": datetime.now().strftime("%Y-%m-%d %H:%M")}
    except Exception as e:
        _crawl_result = {"ok": False, "trigger": trigger, "error": str(e)}
    finally:
        _crawl_in_progress = False
        _crawl_lock.release()

def _latest_added_date() -> str:
    """读 news_data.json 中最新 added_date，用于启动补偿判断。"""
    try:
        with open(os.path.join(BASE_DIR, "news_data.json"), encoding="utf-8") as f:
            return max((e.get("added_date", "") for e in json.load(f)), default="")
    except (IOError, json.JSONDecodeError):
        return ""

def _scheduler_loop() -> None:
    schedule_lib.every().saturday.at("09:00").do(_safe_run_crawl, trigger="scheduled")
    while True:
        schedule_lib.run_pending()
        time.sleep(60)

def _next_saturday_nine() -> str:
    now = datetime.now()
    days_ahead = (5 - now.weekday()) % 7          # 周六=5
    nxt = now + timedelta(days=days_ahead)
    if days_ahead == 0 and now.hour >= 9:
        nxt += timedelta(days=7)
    return nxt.replace(hour=9, minute=0, second=0).strftime("%Y-%m-%d %H:%M")
```

路由保持 `/`、`/crawl`（POST，改用 `_crawl_lock` 互斥，冲突返回 429）、`/status`（新增 `next_run: _next_saturday_nine()` 和 `last_result`）。

`if __name__ == "__main__":` 部分：
1. 启动调度线程（daemon）：`threading.Thread(target=_scheduler_loop, daemon=True).start()`
2. **启动补偿**：若 `_latest_added_date()` 距今 > 7 天，60 秒后 `_safe_run_crawl("startup")`
3. `app.run(host="0.0.0.0", port=5000, debug=False)`（保留 threaded 默认，Flask 1.0+ 默认多线程）

```python
"""多级网页抓取：对 top N 条目进入原文页提取正文/首图/PDF 链接。"""
import os, re, time, hashlib, logging
from datetime import datetime, timedelta
from typing import Dict, List

import trafilatura

from crawler.fetch import build_session, load_config
from crawler.sources import RSS_SOURCES

logger = logging.getLogger("crawler")

BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
CACHE_DIR = os.path.join(BASE_DIR, "cache")
CACHE_TTL_DAYS = 7
DETAIL_TEXT_LIMIT = 2000
PAGE_TIMEOUT = 15

_SOURCE_WEIGHT = {s["name"]: s.get("weight", 1) for s in RSS_SOURCES}
_last_fetch_by_domain: Dict[str, float] = {}

def _domain_throttle(url: str, min_gap: float = 2.0) -> None:
    """同域名两次请求间隔 >= min_gap 秒。"""
    from urllib.parse import urlparse
    domain = urlparse(url).netloc
    wait = _last_fetch_by_domain.get(domain, 0) + min_gap - time.time()
    if wait > 0:
        time.sleep(wait)
    _last_fetch_by_domain[domain] = time.time()

def _cache_path(link: str) -> str:
    return os.path.join(CACHE_DIR, hashlib.md5(link.encode()).hexdigest() + ".html")

def cache_is_fresh(path: str) -> bool:
    if not os.path.exists(path):
        return False
    age = datetime.now() - datetime.fromtimestamp(os.path.getmtime(path))
    return age < timedelta(days=CACHE_TTL_DAYS)

def select_for_detail(entries: List[Dict], top_n: int) -> List[Dict]:
    """每分类按 weight × 新鲜度 选 top_n 条进入详情抓取。"""
    picked: List[Dict] = []
    for cat in ("world", "science", "ai"):
        pool = [e for e in entries if e.get("category") == cat]
        pool.sort(key=lambda e: (_SOURCE_WEIGHT.get(e.get("source"), 1),
                                 e.get("published", "")), reverse=True)
        picked.extend(pool[:top_n])
    return picked

def fetch_detail(entry: Dict, proxy: str) -> Dict:
    """抓取原文页，返回 {"fetched", "text", "image", "pdf_links"}。"""
    os.makedirs(CACHE_DIR, exist_ok=True)
    link = entry["link"]
    cpath = _cache_path(link)
    if cache_is_fresh(cpath):
        with open(cpath, encoding="utf-8") as f:
            html = f.read()
    else:
        needs_proxy = any(s["name"] == entry.get("source") and s.get("needs_proxy")
                          for s in RSS_SOURCES)
        s = build_session(needs_proxy, proxy)
        _domain_throttle(link)
        r = s.get(link, timeout=PAGE_TIMEOUT)
        r.raise_for_status()
        html = r.text
        with open(cpath, "w", encoding="utf-8") as f:
            f.write(html)

    text = trafilatura.extract(html, include_images=True,
                               include_links=True, favor_recall=True) or ""
    first_img = ""
    m = re.search(r'<img[^>]+src=["\']([^"\']+)["\']', html)
    if m:
        first_img = m.group(1)
    pdf_links = sorted(set(re.findall(r'href=["\']([^"\']+\.pdf[^"\']*)["\']', html,
                                      re.IGNORECASE)))
    # arXiv 特判: abs 页 → pdf
    if "arxiv.org/abs/" in link:
        pdf_links = [link.replace("/abs/", "/pdf/")]
    return {"fetched": True, "text": text[:DETAIL_TEXT_LIMIT],
            "image": first_img, "pdf_links": pdf_links}

def enrich(entries: List[Dict]) -> List[Dict]:
    """pipeline 调用入口：给 top N 条目挂 detail 字段。失败静默降级。"""
    cfg = load_config()
    dc = cfg.get("detail_crawl", {})
    if not dc.get("enabled", True):
        return entries
    proxy = cfg.get("proxy", "")
    for entry in select_for_detail(entries, int(dc.get("top_n", 10))):
        try:
            entry["detail"] = fetch_detail(entry, proxy)
        except Exception as e:
            logger.warning("详情抓取失败(%s): %s", entry.get("source"), e)
            entry["detail"] = {"fetched": False}
    return entries
```

```python
# -*- coding: utf-8 -*-
"""PDF 等附件下载归档到 attachments/YYYY-MM/。"""
import os, re, time, logging
from datetime import datetime
from typing import Dict, List

from crawler.fetch import build_session, load_config

logger = logging.getLogger("crawler")

BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
ATTACH_DIR = os.path.join(BASE_DIR, "attachments")

_PDF_NAME_RE = re.compile(r'[^/\\?=]+\.pdf', re.IGNORECASE)

def _filename_from_url(url: str) -> str:
    m = _PDF_NAME_RE.search(url)
    base = m.group(0) if m else "file.pdf"
    return base.split("?")[0][:120]

def download_all(entry: Dict, proxy: str) -> List[Dict]:
    """下载 entry["detail"]["pdf_links"] 中的 PDF，返回附件元数据列表。"""
    cfg = load_config()
    ac = cfg.get("attachment", {})
    max_bytes = int(ac.get("max_mb", 20)) * 1024 * 1024
    needs_proxy = entry.get("source") in {s["name"] for s in
                                          _proxy_source_names()}
    session = build_session(needs_proxy, proxy)
    saved: List[Dict] = []
    for url in entry.get("detail", {}).get("pdf_links", [])[:5]:  # 单条最多 5 个附件
        try:
            head = session.head(url, timeout=15, allow_redirects=True)
            size = int(head.headers.get("Content-Length", 0))
            if size > max_bytes:
                logger.info("附件过大跳过 (%d B): %s", size, url)
                continue
            r = session.get(url, timeout=60)
            r.raise_for_status()
            month_dir = os.path.join(ATTACH_DIR, datetime.now().strftime("%Y-%m"))
            os.makedirs(month_dir, exist_ok=True)
            fname = _filename_from_url(url)
            fpath = os.path.join(month_dir, fname)
            with open(fpath, "wb") as f:
                f.write(r.content)
            saved.append({"filename": fname, "size": len(r.content), "url": url})
            time.sleep(1)
        except Exception as e:
            logger.warning("附件下载失败(%s): %s", url, e)
    return saved

def _proxy_source_names():
    from crawler.sources import RSS_SOURCES
    return [s for s in RSS_SOURCES if s.get("needs_proxy")]
```
